\begin{frame}
  \frametitle{Гиперграфы отношений и их изоморфизм}
  \begin{itemize}
    \item Рёбра: тип~(1) $p^k \to (\omega_{i_1}, \dots, \omega_{i_k})$~--- кортеж, тип~(2) $p^{c,k} \to \{\omega_{i_1}, \dots, \omega_{i_k}\}$~--- множество
  \end{itemize}
  \begin{rudefinition}
    $G_1$ и $G_2$ изоморфны, если есть гиперграф $H$ с переменными вместо имён объектных вершин и подстановки (\emph{унификаторы}) $\lambda_1$, $\lambda_2$: \ $H\lambda_1 = G_1$, $H\lambda_2 = G_2$
  \end{rudefinition}
  \begin{itemize}
    \item $H$ получают из $G_2$ заменой констант переменными и ищут биекцию $\lambda$ с $H\lambda = G_1$
    \item \emph{Характеристика} вершины: вектор $(n_1, \dots, n_k)$ вхождений по местам (тип~1) или число вхождений (тип~2); вершины с равной характеристикой образуют классы
    \item \emph{Структурное отображение} сопоставляет классы переменных и констант: перебор $\prod_k t_k!$ вместо $n!$ \\
          {\small например, $p^3$ на 6 вершинах: $\{[x_5]:[a_4],\ [x_3]:[a_3],\ [x_2, x_6]:[a_1, a_5],\ [x_1, x_4]:[a_2, a_6]\}$~--- 4 подстановки вместо 720}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Алгоритм \ISOMn{}}
  \centering
  \begin{tikzpicture}[node distance=3mm and 6mm, every node/.append style={font=\footnotesize}]
    \node[block] (map) {\proc{MAP}};
    \node[block, right=of map] (best) {\proc{BESTMAP}};
    \node[block, right=of best] (isom) {\ISOMm{}};
    \node[block, right=of isom] (cons) {\proc{CONSISTENCY}};
    \node[block, right=of cons] (red) {\proc{REDUCTION}};
    \node[term, right=of red] (out) {унификатор};
    \node[term, below=12mm of $(best)!0.5!(cons)$] (queue) {очередь вершин $\langle G, H, \lambda \rangle$ по возрастанию $\prod t_k!$, в начале~--- $\langle G, H, \varnothing \rangle$};

    \draw[arr] (map) -- (best);
    \draw[arr] (best) -- (isom);
    \draw[arr] (isom) -- (cons);
    \draw[arr] (cons) -- (red);
    \draw[arr] (red) -- (out);
    \draw[arr] (map.south |- queue.north) -- (map.south);
    \draw[arr] (red.south) |- node[pos=0.3, right, font=\scriptsize] {частичные} (queue.east);
    \draw[arr, densely dashed] (isom.south) -- node[pos=0.45, left, font=\scriptsize] {$r = 0$} (isom.south |- queue.north);
    \draw[arr, densely dashed] (cons.south) -- node[pos=0.45, right, font=\scriptsize] {$T = 0$} (cons.south |- queue.north);
  \end{tikzpicture}

  \vspace{8pt}
  \raggedright\small
  \begin{itemize}
    \item \proc{MAP}: структурные отображения для подграфов $G^i$, $H^i\lambda$, где остались переменные
    \item \proc{BESTMAP}: подграф $i_0$ с наименьшим числом возможных подстановок $\prod t_k!$
    \item \ISOMm{}: локальные унификаторы $G^{i_0}$ и $H^{i_0}\lambda$ (\ISOMone{} без шагов 1--3)
    \item \proc{CONSISTENCY}: только те, что вместе с $\lambda$ образуют биекцию
    \item \proc{REDUCTION}: подграфы без переменных сравниваются с $G$; если переменных не осталось~--- унификатор
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Корректность}
  \begin{rulemma}
    Если каждое ребро $H^i\lambda$ проверять один раз~--- после класса, на котором в нём не остаётся переменных,~--- \ISOMm{} отсекает те же подстановки, что и при проверке всех рёбер после каждого класса.
  \end{rulemma}
  \begin{rulemma}
    Если \proc{BESTMAP} выбирает подграф однозначно, все вершины дерева \ISOMn{} и все найденные унификаторы различны: \enquote{склейка} вершин не нужна.
  \end{rulemma}
  \textbf{Проверка реализаций}
  \begin{itemize}
    \item Полный перебор всех $n!$ биекций на малых задачах (обе реализации, $n \le 8$)
    \item Сверка Python и \Cpp{}: ответы, множества изоморфизмов, статистика деревьев, журналы
    \item \NCases{}~задач, \NChecks{}~проверок~--- \textbf{расхождений нет}
  \end{itemize}
\end{frame}
